// ---------- NTC ----------
float adcRawToCelsius(int raw){
  raw = constrain(raw,1,ADC_MAX-1);
  float v = (raw*VREF)/ADC_MAX; v = clampT(v,0.001f,VREF-0.001f);
  float r = (v*THERM_R_FIXED)/(VREF-v); r = max(r,100.0f);
  float invT = (1.0/THERM_T0_K) + (1.0/THERM_BETA)*log(r/THERM_R0);
  return (1.0/invT) - 273.15;
}
float readNTC_Celsius_multi(int n){
  const int N=min(n,15); int buf[15];
  for(int i=0;i<N;i++){ buf[i]=analogRead(THERM_ADC_PIN); delayMicroseconds(250); }
  // median + trimmed mean
  for(int i=0;i<N;i++) for(int j=i+1;j<N;j++) if(buf[j]<buf[i]){int t=buf[i];buf[i]=buf[j];buf[j]=t;}
  int m=buf[N/2]; long sum=0; int cnt=0;
  for(int i=0;i<N;i++){ if(abs(buf[i]-m)<=100){ sum+=buf[i]; cnt++; } }
  int rawAvg = cnt? (int)(sum/cnt) : m;
  return adcRawToCelsius(rawAvg);
}

}

// ---------- Config (Preferences) ----------
void loadConfig(){
  prefs.begin("bbtemp",true);
  cfg.offsetC = prefs.getFloat("ofsC",0.0);
  cfg.lowC    = clampT(prefs.getFloat("lowC",2.0f),-20.0f,40.0f);
  cfg.highC   = clampT(prefs.getFloat("highC",6.0f),-20.0f,40.0f);
  if(cfg.highC<=cfg.lowC){cfg.lowC=2.0; cfg.highC=6.0;}
  cfg.phones  = prefs.getString("phones","");
  cfg.soundEnable = prefs.getBool("sound",true);
  cfg.callEnable  = prefs.getBool("call",false);
  cfg.smsIntervalMin = clampT((uint16_t)prefs.getUShort("smsMin",5), (uint16_t)1, (uint16_t)1440);
  prefs.end();
}

// ---------- GSM (SMS/Call + time) ----------
void gsmSend(const String&s){ GSM.print(s); }
bool waitForChar(char ch,uint32_t ms=7000){
  uint32_t t=millis();
  while(millis()-t<ms){ while(GSM.available()){ if(GSM.read()==ch) return true; } }
  return false;
}
String gsmReadAll(uint32_t ms=200){
  uint32_t t=millis(); String r;
  while(millis()-t<ms){ while(GSM.available()){ r += (char)GSM.read(); } }
  return r;
}
void gsmInit(){
  GSM.begin(GSM_BAUD,SERIAL_8N1,GSM_RX_PIN,GSM_TX_PIN);
  delay(400); GSM.setTimeout(2000);
  gsmSend("AT\r"); delay(150); gsmReadAll();
  gsmSend("ATE0\r"); delay(150); gsmReadAll();
  gsmSend("AT+CMGF=1\r"); delay(150); gsmReadAll();
  gsmSend("AT+CSCS=\"GSM\"\r"); delay(150); gsmReadAll();
  gsmSend("AT+CLTS=1\r"); delay(200); gsmReadAll(); // network time update if supported
}

// Try to refresh lastTimeStr using +CCLK?
void gsmPollTime(){
  if(millis()-lastGsmTimePoll < GSM_TIME_POLL_MS) return;
  lastGsmTimePoll = millis();
  gsmSend("AT+CCLK?\r");
  delay(200);
  String r = gsmReadAll(400);
  int p = r.indexOf("+CCLK: ");
  if(p>=0){
    int q = r.indexOf('\n', p);
    String line = (q>p) ? r.substring(p, q) : r.substring(p);
    // Expect: +CCLK: "yy/MM/dd,HH:MM:SS+TZ"
    int q1 = line.indexOf('"'), q2 = line.indexOf('"', q1+1);
    if(q1>=0 && q2>q1){
      String ts = line.substring(q1+1, q2);
      if(ts.length() >= 17){
        String yy=ts.substring(0,2), MM=ts.substring(3,5), dd=ts.substring(6,8);
        String hh=ts.substring(9,11), mi=ts.substring(12,14), ss=ts.substring(15,17);
        lastTimeStr = String("20")+yy+"-"+MM+"-"+dd+" "+hh+":"+mi+":"+ss;
      } else {
        lastTimeStr = ts; // fallback raw
      }
    }
  }
}

void sendSMS_one(const String&to,const String&body){
  if(to.length()<3) return;
  gsmSend("AT+CMGS=\""+to+"\"\r");
  if(!waitForChar('>',8000)){ lastSmsInfo="no-prompt"; return; }
  gsmSend(body); GSM.write(26);
  lastSmsInfo="sent";
}
void broadcastSMS(const String&body){
  if(!cfg.phones.length()) return;
  int i=0; while(i<cfg.phones.length()){
    int c=cfg.phones.indexOf(',',i);
    String num=(c==-1)?cfg.phones.substring(i):cfg.phones.substring(i,c);
    num.trim(); if(num.length()>=3) sendSMS_one(num,body);
    if(c==-1) break; i=c+1;
  }
}
String firstPhone(){ String s=cfg.phones; s.trim(); int c=s.indexOf(','); if(c>0) s=s.substring(0,c); s.trim(); return s; }
void placeCallOnce(){
  if(!cfg.callEnable){ lastCallInfo="off"; return; }
  String n=firstPhone(); if(n.length()<3){ lastCallInfo="no-num"; return; }
  gsmSend("ATD"+n+";\r"); lastCallInfo="dialing";
  uint32_t t0=millis(); while(millis()-t0<20000) delay(10);
  gsmSend("ATH\r"); lastCallInfo="called";
}

String uptimeHMS(uint32_t ms){
  uint32_t s=ms/1000, h=s/3600, m=(s%3600)/60, ss=s%60;
  char b[16]; snprintf(b,sizeof(b),"%02u:%02u:%02u",(unsigned)h,(unsigned)m,(unsigned)ss);
  return String(b);
}
String currentTimestampLabel(){
  if(lastTimeStr.length()) return lastTimeStr;        // GSM time if available
  return String("UP ")+uptimeHMS(millis());           // else uptime
}

void handleSaveCfg(){
  if(server.method()!=HTTP_POST){server.send(405,"text/plain","POST only");return;}
  if(server.hasArg("ofs"))    cfg.offsetC=clampT(server.arg("ofs").toFloat(),-5.0f,5.0f);
  if(server.hasArg("low"))    cfg.lowC   =clampT(server.arg("low").toFloat(),-20.0f,40.0f);
  if(server.hasArg("high"))   cfg.highC  =clampT(server.arg("high").toFloat(),-20.0f,40.0f);
  if(cfg.highC<=cfg.lowC){cfg.lowC=2.0; cfg.highC=6.0;}
  if(server.hasArg("phones")) cfg.phones = server.arg("phones");
  cfg.soundEnable = server.hasArg("sound");
  cfg.callEnable  = server.hasArg("call");
  if(server.hasArg("smsmin")){
    int v = server.arg("smsmin").toInt();
    cfg.smsIntervalMin = clampT((uint16_t)(v>0?v:5), (uint16_t)1, (uint16_t)1440);
  }
  saveConfig(); server.send(200,"text/plain","ok");
}

// ---------- Alarm logic ----------
inline bool outOfRange(float t){return t<cfg.lowC || t>cfg.highC;}

void buzzerPatternUpdate(){
  static bool hi=false; static uint32_t toggleT=0; const uint32_t now=millis();
  if(!cfg.soundEnable){ ledcWrite(LEDC_CH,0); digitalWrite(RELAY_PIN,(alarmState==ALARM)?HIGH:LOW); return; }
  if(alarmState==PREALARM){
    // Gentle chirp while pre-alarm
    const uint32_t period=1500,on=220;
    if((now%period)<on){ ledcWriteTone(LEDC_CH,1000); ledcWrite(LEDC_CH,LEDC_DUTY);}
    else ledcWrite(LEDC_CH,0);
    digitalWrite(RELAY_PIN,LOW);
  } else if(alarmState==ALARM){
    // Fire-alarm whoop
    if(now-toggleT>=500){ toggleT=now; hi=!hi; ledcWriteTone(LEDC_CH,hi?880:660); ledcWrite(LEDC_CH,LEDC_DUTY); }
    digitalWrite(RELAY_PIN,HIGH);
  } else {
    ledcWrite(LEDC_CH,0); digitalWrite(RELAY_PIN,LOW);
  }
}

void enterState(AlarmState s){
  alarmState=s;
  if(s==ALARM){
    String msg="ALERT: Temp="+String(tempFilt,2)+"C (range "+String(cfg.lowC,1)+"-"+String(cfg.highC,1)+"C)";
    broadcastSMS(msg); lastSmsMs=millis();
    if(cfg.callEnable){ placeCallOnce(); lastCallMs=millis(); }
  }
}

void setup(){
  Serial.begin(115200);
  pinMode(RELAY_PIN,OUTPUT); digitalWrite(RELAY_PIN,LOW);
  pinMode(BUZZER_PIN,OUTPUT);
  ledcSetup(LEDC_CH,1000,LEDC_RES); ledcAttachPin(BUZZER_PIN,LEDC_CH); ledcWrite(LEDC_CH,0);

  analogReadResolution(12);

  Wire.begin(); lcd.init(); lcd.backlight(); lcdSplash();
  loadConfig(); fsInit(); openLogAppend(); rotateLogIfNeeded();

  WiFi.mode(WIFI_AP); WiFi.softAP(AP_SSID,AP_PASS);

  GSM.begin(GSM_BAUD,SERIAL_8N1,GSM_RX_PIN,GSM_TX_PIN); delay(300); gsmInit();

  server.on("/",            [](){ server.send(200,"text/html",htmlDash()); });
  server.on("/settings",    [](){ server.send(200,"text/html",htmlSettings()); });
  server.on("/history",     handleHistory);
  server.on("/api/info",    handleInfo);
  server.on("/api/getcfg",  handleGetCfg);
  server.on("/api/savecfg", HTTP_POST, handleSaveCfg);
  server.on("/api/testsms", handleTestSMS);
  server.on("/api/sound",   handleSound);
  server.on("/log.csv",     handleLogCsv);
  server.on("/export",      handleExport);
  // Maintenance
  server.on("/api/resetminmax", handleResetMinMax);
  server.on("/api/clearlog",    handleClearLog);
  server.on("/api/factoryreset",handleFactoryReset);

  server.onNotFound([](){ server.send(404,"text/plain","Not found"); });
  server.begin();

  // initial temperature & smoothing
  float t0=readNTC_Celsius_multi(PER_READ_SAMPLES)+cfg.offsetC;
  for(int i=0;i<AVG_N;i++) avgBuf[i]=t0; avgFill=AVG_N;
  tempFilt=minC=maxC=t0; lcd.clear();
}

void loop(){
  server.handleClient();

  // Keep trying to fetch GSM time (for timestamps)
  gsmPollTime();

  if(millis()-lastSample>=SAMPLE_MS){
    lastSample=millis();
    float t=readNTC_Celsius_multi(PER_READ_SAMPLES)+cfg.offsetC;
    tempFilt=filtTempMA(t);

    if(isnan(minC)||tempFilt<minC) minC=tempFilt;
    if(isnan(maxC)||tempFilt>maxC) maxC=tempFilt;

    bool out=outOfRange(tempFilt);
    if(out){ outCnt++; inCnt=0; } else { inCnt++; outCnt=0; }

    if(alarmState==NORMAL   && outCnt>=PREALARM_SEC) enterState(PREALARM);
    if(alarmState==PREALARM && outCnt>=ALARM_SEC)   enterState(ALARM);
    if((alarmState==PREALARM||alarmState==ALARM) && inCnt>=CLEAR_SEC){
      alarmState=NORMAL; lastSmsInfo="cleared"; lastCallInfo="cleared";
    }

    // Repeat SMS while alarm is ON per user-set interval
    if(alarmState==ALARM){
      uint32_t interval_ms = (uint32_t)cfg.smsIntervalMin * 60UL * 1000UL;
      if(millis()-lastSmsMs >= interval_ms){
        broadcastSMS("REMINDER: Temp="+String(tempFilt,2)+"C outside "+String(cfg.lowC,1)+"-"+String(cfg.highC,1)+"C");
        lastSmsMs=millis();
      }
    }

    lcdScreen();

    if(millis()-lastLogWrite>=LOG_WRITE_MS){
      lastLogWrite=millis(); rotateLogIfNeeded(); appendLogRow(tempFilt,(alarmState==ALARM));
    }
  }

  buzzerPatternUpdate();
}
